91.	اَمَّا مَنِ اسْتَغْنٰى⦿
92.	فَاَنْتَ لَهٗ تَصَدّٰى⦿
93.	وَ مَا عَلَیْكَ اَلَّا یَزَّكّٰى⦿
94.	وَ اَمَّا مَنْ جَآءَكَ یَسْعٰى⦿
95.	وَ هُوَ یَخْشٰى⦿
96.	فَاَنْتَ عَنْهُ تَلَهّٰى⦿
97.	كَلَّاۤ اِنَّهَا تَذْكِرَةٌ⦿
98.	فَمَنْ شَآءَ ذَكَرَهٗ⦿
99.	فِیْ صُحُفٍ مُّكَرَّمَةٍ⦿
100.	مَّرْفُوْعَةٍ مُّطَهَّرَةٍ⦿
101.	بِاَیْدِیْ سَفَرَةٍ⦿
102.	كِرَامٍۭ بَرَرَةٍ⦿
103.	قُتِلَ الْاِنْسَانُ مَاۤ اَكْفَرَهٗ⦿
104.	مِنْ اَیِّ شَیْءٍ خَلَقَهٗ⦿
105.	مِنْ نُّطْفَةٍؕ-خَلَقَهٗ فَقَدَّرَهٗ⦿
106.	ثُمَّ السَّبِیْلَ یَسَّرَهٗ⦿
107.	ثُمَّ اَمَاتَهٗ فَاَقْبَرَهٗ⦿
108.	ثُمَّ اِذَا شَآءَ اَنْشَرَهٗ⦿
109.	كَلَّا لَمَّا یَقْضِ مَاۤ اَمَرَهٗ⦿
110.	فَلْیَنْظُرِ الْاِنْسَانُ اِلٰى طَعَامِهٖ⦿
111.	اَنَّا صَبَبْنَا الْمَآءَ صَبًّا⦿
112.	ثُمَّ شَقَقْنَا الْاَرْضَ شَقًّا⦿
113.	فَاَنْۢبَتْنَا فِیْهَا حَبًّا⦿
114.	وَّ عِنَبًا وَّ قَضْبًا⦿
115.	وَّ زَیْتُوْنًا وَّ نَخْلًا⦿
116.	وَّ حَدَآىٕقَ غُلْبًا⦿
117.	وَّ فَاكِهَةً وَّ اَبًّا⦿
118.	مَّتَاعًا لَّكُمْ وَ لِاَنْعَامِكُمْ⦿
119.	فَاِذَا جَآءَتِ الصَّآخَّةُ⦿
120.	یَوْمَ یَفِرُّ الْمَرْءُ مِنْ اَخِیْهِ⦿
121.	وَ اُمِّهٖ وَ اَبِیْهِ⦿
122.	وَ صَاحِبَتِهٖ وَ بَنِیْهِ⦿
123.	لِكُلِّ امْرِئٍ مِّنْهُمْ یَوْمَىٕذٍ شَاْنٌ یُّغْنِیْهِ⦿
124.	وُجُوْهٌ یَّوْمَىٕذٍ مُّسْفِرَةٌ⦿
125.	ضَاحِكَةٌ مُّسْتَبْشِرَةٌ⦿
126.	وَ وُجُوْهٌ یَّوْمَىٕذٍ عَلَیْهَا غَبَرَةٌ⦿
127.	تَرْهَقُهَا قَتَرَةٌ⦿
128.	اُولٰٓىٕكَ هُمُ الْكَفَرَةُ الْفَجَرَةُ⦿
سُورَۃُ التَّکوِیرِ
129.	اِذَا الشَّمْسُ كُوِّرَتْ⦿
130.	وَ اِذَا النُّجُوْمُ انْكَدَرَتْ⦿
131.	وَ اِذَا الْجِبَالُ سُیِّرَتْ⦿
132.	وَ اِذَا الْعِشَارُ عُطِّلَتْ⦿
133.	وَ اِذَا الْوُحُوْشُ حُشِرَتْ⦿
134.	وَ اِذَا الْبِحَارُ سُجِّرَتْ⦿
135.	وَ اِذَا النُّفُوْسُ زُوِّجَتْ⦿
136.	وَ اِذَا الْمَوْءٗدَةُ سُىٕلَتْ⦿
137.	بِاَیِّ ذَنْۢبٍ قُتِلَتْ⦿
138.	وَ اِذَا الصُّحُفُ نُشِرَتْ⦿
139.	وَ اِذَا السَّمَآءُ كُشِطَتْ⦿
140.	وَ اِذَا الْجَحِیْمُ سُعِّرَتْ⦿
141.	وَ اِذَا الْجَنَّةُ اُزْلِفَتْ⦿
142.	عَلِمَتْ نَفْسٌ مَّاۤ اَحْضَرَتْ⦿
143.	فَلَاۤ اُقْسِمُ بِالْخُنَّسِ⦿
144.	الْجَوَارِ الْكُنَّسِ⦿
145.	وَ الَّیْلِ اِذَا عَسْعَسَ⦿
146.	وَ الصُّبْحِ اِذَا تَنَفَّسَ⦿
147.	اِنَّهٗ لَقَوْلُ رَسُوْلٍ كَرِیْمٍ⦿
148.	ذِیْ قُوَّةٍ عِنْدَ ذِی الْعَرْشِ مَكِیْنٍ⦿
149.	مُّطَاعٍ ثَمَّ اَمِیْنٍ⦿
150.	وَ مَا صَاحِبُكُمْ بِمَجْنُوْنٍ⦿
151.	وَ لَقَدْ رَاٰهُ بِالْاُفُقِ الْمُبِیْنِ⦿
152.	وَ مَا هُوَ عَلَى الْغَیْبِ بِضَنِیْنٍ⦿
153.	وَ مَا هُوَ بِقَوْلِ شَیْطٰنٍ رَّجِیْمٍ⦿
154.	فَاَیْنَ تَذْهَبُوْنَ⦿
155.	اِنْ هُوَ اِلَّا ذِكْرٌ لِّلْعٰلَمِیْنَ⦿
156.	لِمَنْ شَآءَ مِنْكُمْ اَنْ یَّسْتَقِیْمَ⦿
157.	وَ مَا تَشَآءُوْنَ اِلَّاۤ اَنْ یَّشَآءَ اللّٰهُ رَبُّ الْعٰلَمِیْنَ⦿
سُورَۃُ الاِنفِطَارِ
158.	اِذَا السَّمَآءُ انْفَطَرَتْ⦿
159.	وَ اِذَا الْكَوَاكِبُ انْتَثَرَتْ⦿
160.	وَ اِذَا الْبِحَارُ فُجِّرَتْ⦿
161.	وَ اِذَا الْقُبُوْرُ بُعْثِرَتْ⦿
162.	عَلِمَتْ نَفْسٌ مَّا قَدَّمَتْ وَ اَخَّرَتْ⦿
163.	یٰۤاَیُّهَا الْاِنْسَانُ مَا غَرَّكَ بِرَبِّكَ الْكَرِیْمِ⦿
164.	الَّذِیْ خَلَقَكَ فَسَوّٰىكَ فَعَدَلَكَ⦿
165.	فِیْۤ اَیِّ صُوْرَةٍ مَّا شَآءَ رَكَّبَكَ⦿
166.	كَلَّا بَلْ تُكَذِّبُوْنَ بِالدِّیْنِ⦿
167.	وَ اِنَّ عَلَیْكُمْ لَحٰفِظِیْنَ⦿
168.	كِرَامًا كَاتِبِیْنَ⦿
169.	یَعْلَمُوْنَ مَا تَفْعَلُوْنَ⦿
170.	اِنَّ الْاَبْرَارَ لَفِیْ نَعِیْمٍ⦿
171.	وَ اِنَّ الْفُجَّارَ لَفِیْ جَحِیْمٍ⦿
172.	یَّصْلَوْنَهَا یَوْمَ الدِّیْنِ⦿
173.	وَ مَا هُمْ عَنْهَا بِغَآىٕبِیْنَ⦿
174.	وَ مَاۤ اَدْرٰىكَ مَا یَوْمُ الدِّیْنِ⦿
175.	ثُمَّ مَاۤ اَدْرٰىكَ مَا یَوْمُ الدِّیْنِ⦿
176.	یَوْمَ لَا تَمْلِكُ نَفْسٌ لِّنَفْسٍ شَیْــٴًـاؕ-وَ الْاَمْرُ یَوْمَىٕذٍ لِّلّٰهِ⦿
سُورَۃُ المُطَفِّفِینَ
177.	وَیْلٌ لِّلْمُطَفِّفِیْنَ⦿
178.	الَّذِیْنَ اِذَا اكْتَالُوْا عَلَى النَّاسِ یَسْتَوْفُوْنَ⦿
179.	وَ اِذَا كَالُوْهُمْ اَوْ وَّ زَنُوْهُمْ یُخْسِرُوْنَ⦿
180.	اَلَا یَظُنُّ اُولٰٓىٕكَ اَنَّهُمْ مَّبْعُوْثُوْنَ⦿
181.	لِیَوْمٍ عَظِیْمٍ⦿
182.	یَّوْمَ یَقُوْمُ النَّاسُ لِرَبِّ الْعٰلَمِیْنَ⦿
183.	كَلَّاۤ اِنَّ كِتٰبَ الْفُجَّارِ لَفِیْ سِجِّیْنٍ⦿
184.	وَ مَاۤ اَدْرٰىكَ مَا سِجِّیْنٌ⦿
185.	كِتٰبٌ مَّرْقُوْمٌ⦿
186.	وَیْلٌ یَّوْمَىٕذٍ لِّلْمُكَذِّبِیْنَ⦿
187.	الَّذِیْنَ یُكَذِّبُوْنَ بِیَوْمِ الدِّیْنِ⦿
188.	وَ مَا یُكَذِّبُ بِهٖۤ اِلَّا كُلُّ مُعْتَدٍ اَثِیْمٍ⦿
189.	اِذَا تُتْلٰى عَلَیْهِ اٰیٰتُنَا قَالَ اَسَاطِیْرُ الْاَوَّلِیْنَ⦿
190.	كَلَّا بَلْٚ- رَانَ عَلٰى قُلُوْبِهِمْ مَّا كَانُوْا یَكْسِبُوْنَ⦿
191.	كَلَّاۤ اِنَّهُمْ عَنْ رَّبِّهِمْ یَوْمَىٕذٍ لَّمَحْجُوْبُوْنَ⦿
192.	ثُمَّ اِنَّهُمْ لَصَالُوا الْجَحِیْمِ⦿
193.	ثُمَّ یُقَالُ هٰذَا الَّذِیْ كُنْتُمْ بِهٖ تُكَذِّبُوْنَ⦿
194.	كَلَّاۤ اِنَّ كِتٰبَ الْاَبْرَارِ لَفِیْ عِلِّیِّیْنَ⦿
195.	وَ مَاۤ اَدْرٰىكَ مَا عِلِّیُّوْنَ⦿
196.	كِتٰبٌ مَّرْقُوْمٌ⦿
197.	یَّشْهَدُهُ الْمُقَرَّبُوْنَ⦿
198.	اِنَّ الْاَبْرَارَ لَفِیْ نَعِیْمٍ⦿
199.	عَلَى الْاَرَآىٕكِ یَنْظُرُوْنَ⦿
200.	تَعْرِفُ فِیْ وُجُوْهِهِمْ نَضْرَةَ النَّعِیْمِ⦿
201.	یُسْقَوْنَ مِنْ رَّحِیْقٍ مَّخْتُوْمٍ⦿
202.	خِتٰمُهٗ مِسْكٌؕ-وَ فِیْ ذٰلِكَ فَلْیَتَنَافَسِ الْمُتَنَافِسُوْنَ⦿
203.	وَ مِزَاجُهٗ مِنْ تَسْنِیْمٍ⦿
204.	عَیْنًا یَّشْرَبُ بِهَا الْمُقَرَّبُوْنَ⦿
205.	اِنَّ الَّذِیْنَ اَجْرَمُوْا كَانُوْا مِنَ الَّذِیْنَ اٰمَنُوْا یَضْحَكُوْنَ⦿
206.	وَ اِذَا انْقَلَبُوْۤا اِلٰۤى اَهْلِهِمُ انْقَلَبُوْا فَـكِهِیْنَ⦿
207.	وَ اِذَا رَاَوْهُمْ قَالُوْۤا اِنَّ هٰۤؤُلَآءِ لَضَآلُّوْنَ⦿
208.	وَ مَاۤ اُرْسِلُوْا عَلَیْهِمْ حٰفِظِیْنَ⦿
209.	فَالْیَوْمَ الَّذِیْنَ اٰمَنُوْا مِنَ الْكُفَّارِ یَضْحَكُوْنَ⦿
210.	عَلَى الْاَرَآىٕكِۙ-یَنْظُرُوْنَ⦿
211.	هَلْ ثُوِّبَ الْكُفَّارُ مَا كَانُوْا یَفْعَلُوْنَ⦿
سُورَۃُ الاِنشِقَاقِ
212.	اِذَا السَّمَآءُ انْشَقَّتْ⦿
213.	وَ اَذِنَتْ لِرَبِّهَا وَ حُقَّتْ⦿
214.	وَ اِذَا الْاَرْضُ مُدَّتْ⦿
215.	وَ اَلْقَتْ مَا فِیْهَا وَ تَخَلَّتْ⦿
216.	وَ اَذِنَتْ لِرَبِّهَا وَ حُقَّتْ⦿
217.	یٰۤاَیُّهَا الْاِنْسَانُ اِنَّكَ كَادِحٌ اِلٰى رَبِّكَ كَدْحًا فَمُلٰقِیْهِ⦿
218.	فَاَمَّا مَنْ اُوْتِیَ كِتٰبَهٗ بِیَمِیْنِهٖ⦿
219.	فَسَوْفَ یُحَاسَبُ حِسَابًا یَّسِیْرًا⦿
220.	وَّ یَنْقَلِبُ اِلٰۤى اَهْلِهٖ مَسْرُوْرًا⦿
221.	وَ اَمَّا مَنْ اُوْتِیَ كِتٰبَهٗ وَرَآءَ ظَهْرِهٖ⦿
222.	فَسَوْفَ یَدْعُوْا ثُبُوْرًا⦿
223.	وَّ یَصْلٰى سَعِیْرًا⦿
224.	اِنَّهٗ كَانَ فِیْۤ اَهْلِهٖ مَسْرُوْرًا⦿
225.	اِنَّهٗ ظَنَّ اَنْ لَّنْ یَّحُوْرَ⦿
226.	بَلٰۤىۚۛ-اِنَّ رَبَّهٗ كَانَ بِهٖ بَصِیْرًا⦿
227.	فَلَاۤ اُقْسِمُ بِالشَّفَقِ⦿
228.	وَ الَّیْلِ وَ مَا وَسَقَ⦿
229.	وَ الْقَمَرِ اِذَا اتَّسَقَ⦿
230.	لَتَرْكَبُنَّ طَبَقًا عَنْ طَبَقٍ⦿
231.	فَمَا لَهُمْ لَا یُؤْمِنُوْنَ⦿
232.	وَ اِذَا قُرِئَ عَلَیْهِمُ الْقُرْاٰنُ لَا یَسْجُدُوْنَ⦿
233.	بَلِ الَّذِیْنَ كَفَرُوْا یُكَذِّبُوْنَ⦿
234.	وَ اللّٰهُ اَعْلَمُ بِمَا یُوْعُوْنَ⦿
235.	فَبَشِّرْهُمْ بِعَذَابٍ اَلِیْمٍ⦿
236.	اِلَّا الَّذِیْنَ اٰمَنُوْا وَ عَمِلُوا الصّٰلِحٰتِ لَهُمْ اَجْرٌ غَیْرُ مَمْنُوْنٍ⦿
سُورَۃُ البُرُوجِ
237.	وَ السَّمَآءِ ذَاتِ الْبُرُوْجِ⦿
238.	وَ الْیَوْمِ الْمَوْعُوْدِ⦿
239.	وَ شَاهِدٍ وَّ مَشْهُوْدٍ⦿
240.	قُتِلَ اَصْحٰبُ الْاُخْدُوْدِ⦿
241.	النَّارِ ذَاتِ الْوَقُوْدِ⦿
242.	اِذْ هُمْ عَلَیْهَا قُعُوْدٌ⦿
243.	وَّ هُمْ عَلٰى مَا یَفْعَلُوْنَ بِالْمُؤْمِنِیْنَ شُهُوْدٌ⦿
244.	وَ مَا نَقَمُوْا مِنْهُمْ اِلَّاۤ اَنْ یُّؤْمِنُوْا بِاللّٰهِ الْعَزِیْزِ الْحَمِیْدِ⦿
245.	الَّذِیْ لَهٗ مُلْكُ السَّمٰوٰتِ وَ الْاَرْضِؕ-وَ اللّٰهُ عَلٰى كُلِّ شَیْءٍ شَهِیْدٌ⦿
246.	اِنَّ الَّذِیْنَ فَتَنُوا الْمُؤْمِنِیْنَ وَ الْمُؤْمِنٰتِ ثُمَّ لَمْ یَتُوْبُوْا فَلَهُمْ عَذَابُ جَهَنَّمَ وَ لَهُمْ عَذَابُ الْحَرِیْقِ⦿
247.	اِنَّ الَّذِیْنَ اٰمَنُوْا وَ عَمِلُوا الصّٰلِحٰتِ لَهُمْ جَنّٰتٌ تَجْرِیْ مِنْ تَحْتِهَا الْاَنْهٰرُ ﲜ ذٰلِكَ الْفَوْزُ الْكَبِیْرُ⦿
248.	اِنَّ بَطْشَ رَبِّكَ لَشَدِیْدٌ⦿
249.	اِنَّهٗ هُوَ یُبْدِئُ وَ یُعِیْدُ⦿
250.	وَ هُوَ الْغَفُوْرُ الْوَدُوْدُ⦿
251.	ذُو الْعَرْشِ الْمَجِیْدُ⦿
252.	فَعَّالٌ لِّمَا یُرِیْدُ⦿
253.	هَلْ اَتٰىكَ حَدِیْثُ الْجُنُوْدِ⦿
254.	فِرْعَوْنَ وَ ثَمُوْدَ⦿
255.	بَلِ الَّذِیْنَ كَفَرُوْا فِیْ تَكْذِیْبٍ⦿
256.	وَّ اللّٰهُ مِنْ وَّرَآىٕهِمْ مُّحِیْطٌ⦿
257.	بَلْ هُوَ قُرْاٰنٌ مَّجِیْدٌ⦿
258.	فِیْ لَوْحٍ مَّحْفُوْظٍ⦿
سُورَۃُ الطَّارِقِ
259.	وَ السَّمَآءِ وَ الطَّارِقِ⦿
260.	وَ مَاۤ اَدْرٰىكَ مَا الطَّارِقُ⦿
261.	النَّجْمُ الثَّاقِبُ⦿
262.	اِنْ كُلُّ نَفْسٍ لَّمَّا عَلَیْهَا حَافِظٌ⦿
263.	فَلْیَنْظُرِ الْاِنْسَانُ مِمَّ خُلِقَ⦿
264.	خُلِقَ مِنْ مَّآءٍ دَافِقٍ⦿
265.	یَّخْرُ جُ مِنْۢ بَیْنِ الصُّلْبِ وَ التَّرَآىٕبِ⦿
266.	اِنَّهٗ عَلٰى رَجْعِهٖ لَقَادِرٌ⦿
267.	یَوْمَ تُبْلَى السَّرَآىٕرُ⦿
268.	فَمَا لَهٗ مِنْ قُوَّةٍ وَّ لَا نَاصِرٍ⦿
269.	وَ السَّمَآءِ ذَاتِ الرَّجْعِ⦿
270.	وَ الْاَرْضِ ذَاتِ الصَّدْعِ⦿
271.	اِنَّهٗ لَقَوْلٌ فَصْلٌ⦿
272.	وَّ مَا هُوَ بِالْهَزْلِ⦿
273.	اِنَّهُمْ یَكِیْدُوْنَ كَیْدًا⦿
274.	وَّ اَكِیْدُ كَیْدًا⦿
275.	فَمَهِّلِ الْكٰفِرِیْنَ اَمْهِلْهُمْ رُوَیْدًا⦿
سُورَۃُ الاَعلیٰ
276.	سَبِّحِ اسْمَ رَبِّكَ الْاَعْلَى⦿
277.	الَّذِیْ خَلَقَ فَسَوّٰى⦿
278.	وَ الَّذِیْ قَدَّرَ فَهَدٰى⦿
279.	وَ الَّذِیْۤ اَخْرَ جَ الْمَرْعٰى⦿
280.	فَجَعَلَهٗ غُثَآءً اَحْوٰى⦿
281.	سَنُقْرِئُكَ فَلَا تَنْسٰۤى⦿
282.	اِلَّا مَا شَآءَ اللّٰهُؕ-اِنَّهٗ یَعْلَمُ الْجَهْرَ وَ مَا یَخْفٰى⦿
283.	وَ نُیَسِّرُكَ لِلْیُسْرٰى⦿
284.	فَذَكِّرْ اِنْ نَّفَعَتِ الذِّكْرٰى⦿
285.	سَیَذَّكَّرُ مَنْ یَّخْشٰى⦿
286.	وَ یَتَجَنَّبُهَا الْاَشْقَى⦿
287.	الَّذِیْ یَصْلَى النَّارَ الْكُبْرٰى⦿
288.	ثُمَّ لَا یَمُوْتُ فِیْهَا وَ لَا یَحْیٰى⦿
289.	قَدْ اَفْلَحَ مَنْ تَزَكّٰى⦿
290.	وَ ذَكَرَ اسْمَ رَبِّهٖ فَصَلّٰى⦿
291.	بَلْ تُؤْثِرُوْنَ الْحَیٰوةَ الدُّنْیَا⦿
292.	وَ الْاٰخِرَةُ خَیْرٌ وَّ اَبْقٰى⦿
293.	اِنَّ هٰذَا لَفِی الصُّحُفِ الْاُوْلٰى⦿
294.	صُحُفِ اِبْرٰهِیْمَ وَ مُوْسٰى⦿
سُورَۃُ الغَاشِیَۃِ
295.	هَلْ اَتٰىكَ حَدِیْثُ الْغَاشِیَةِ⦿
296.	وُجُوْهٌ یَّوْمَىٕذٍ خَاشِعَةٌ⦿
297.	عَامِلَةٌ نَّاصِبَةٌ⦿
298.	تَصْلٰى نَارًا حَامِیَةً⦿
299.	تُسْقٰى مِنْ عَیْنٍ اٰنِیَةٍ⦿
300.	لَیْسَ لَهُمْ طَعَامٌ اِلَّا مِنْ ضَرِیْعٍ⦿
301.	لَّا یُسْمِنُ وَ لَا یُغْنِیْ مِنْ جُوْ عٍ⦿
302.	وُجُوْهٌ یَّوْمَىٕذٍ نَّاعِمَةٌ⦿
303.	لِّسَعْیِهَا رَاضِیَةٌ⦿
304.	فِیْ جَنَّةٍ عَالِیَةٍ⦿
305.	لَّا تَسْمَعُ فِیْهَا لَاغِیَةً⦿
306.	فِیْهَا عَیْنٌ جَارِیَةٌ⦿
307.	فِیْهَا سُرُرٌ مَّرْفُوْعَةٌ⦿
308.	وَّ اَكْوَابٌ مَّوْضُوْعَةٌ⦿
309.	وَّ نَمَارِقُ مَصْفُوْفَةٌ⦿
310.	وَّ زَرَابِیُّ مَبْثُوْثَةٌ⦿
311.	اَفَلَا یَنْظُرُوْنَ اِلَى الْاِبِلِ كَیْفَ خُلِقَتْ⦿
312.	وَ اِلَى السَّمَآءِ كَیْفَ رُفِعَتْ⦿
313.	وَ اِلَى الْجِبَالِ كَیْفَ نُصِبَتْ⦿
314.	وَ اِلَى الْاَرْضِ كَیْفَ سُطِحَتْ⦿
315.	فَذَكِّرْ۫ؕ-اِنَّمَاۤ اَنْتَ مُذَكِّرٌ⦿
316.	لَسْتَ عَلَیْهِمْ بِمُصَۜیْطِرٍ⦿
317.	اِلَّا مَنْ تَوَلّٰى وَ كَفَرَ⦿
318.	فَیُعَذِّبُهُ اللّٰهُ الْعَذَابَ الْاَكْبَرَ⦿
319.	اِنَّ اِلَیْنَاۤ اِیَابَهُمْ⦿
320.	ثُمَّ اِنَّ عَلَیْنَا حِسَابَهُمْ⦿
سُورَۃُ الفَجرِ
321.	وَ الْفَجْرِ⦿
322.	وَ لَیَالٍ عَشْرٍ⦿
323.	وَّ الشَّفْعِ وَ الْوَتْرِ⦿
324.	وَ الَّیْلِ اِذَا یَسْرِ⦿
325.	هَلْ فِیْ ذٰلِكَ قَسَمٌ لِّذِیْ حِجْرٍ⦿
326.	اَلَمْ تَرَ كَیْفَ فَعَلَ رَبُّكَ بِعَادٍ⦿
327.	اِرَمَ ذَاتِ الْعِمَادِ⦿
328.	الَّتِیْ لَمْ یُخْلَقْ مِثْلُهَا فِی الْبِلَادِ⦿
329.	وَ ثَمُوْدَ الَّذِیْنَ جَابُوا الصَّخْرَ بِالْوَادِ⦿
330.	وَ فِرْعَوْنَ ذِی الْاَوْتَادِ⦿
331.	الَّذِیْنَ طَغَوْا فِی الْبِلَادِ⦿
332.	فَاَكْثَرُوْا فِیْهَا الْفَسَادَ⦿
333.	فَصَبَّ عَلَیْهِمْ رَبُّكَ سَوْطَ عَذَابٍ⦿
334.	اِنَّ رَبَّكَ لَبِالْمِرْصَادِ⦿
335.	فَاَمَّا الْاِنْسَانُ اِذَا مَا ابْتَلٰىهُ رَبُّهٗ فَاَكْرَمَهٗ وَ نَعَّمَهٗ ﳔ فَیَقُوْلُ رَبِّیْۤ اَكْرَمَنِ⦿
336.	وَ اَمَّاۤ اِذَا مَا ابْتَلٰىهُ فَقَدَرَ عَلَیْهِ رِزْقَهٗ ﳔ فَیَقُوْلُ رَبِّیْۤ اَهَانَنِ⦿
337.	كَلَّا بَلْ لَّا تُكْرِمُوْنَ الْیَتِیْمَ⦿
338.	وَ لَا تَحٰٓضُّوْنَ عَلٰى طَعَامِ الْمِسْكِیْنِ⦿
339.	وَ تَاْكُلُوْنَ التُّرَاثَ اَكْلًا لَّمًّا⦿
340.	وَّ تُحِبُّوْنَ الْمَالَ حُبًّا جَمًّا⦿
341.	كَلَّاۤ اِذَا دُكَّتِ الْاَرْضُ دَكًّا دَكًّا⦿
342.	وَّ جَآءَ رَبُّكَ وَ الْمَلَكُ صَفًّا صَفًّا⦿
343.	وَ جِایْٓءَ یَوْمَىٕذٍۭ بِجَهَنَّمَ یَوْمَىٕذٍ یَّتَذَكَّرُ الْاِنْسَانُ وَ اَنّٰى لَهُ الذِّكْرٰى⦿
344.	یَقُوْلُ یٰلَیْتَنِیْ قَدَّمْتُ لِحَیَاتِیْ⦿
345.	فَیَوْمَىٕذٍ لَّا یُعَذِّبُ عَذَابَهٗۤ اَحَدٌ⦿
346.	وَّ لَا یُوْثِقُ وَ ثَاقَهٗۤ اَحَدٌ⦿
347.	یٰۤاَیَّتُهَا النَّفْسُ الْمُطْمَىٕنَّةُ⦿
348.	ارْجِعِیْۤ اِلٰى رَبِّكِ رَاضِیَةً مَّرْضِیَّةً⦿
349.	فَادْخُلِیْ فِیْ عِبٰدِیْ⦿
350.	وَ ادْخُلِیْ جَنَّتِیْ⦿
سُورَۃُ البَلَدِ
351.	لَاۤ اُقْسِمُ بِهٰذَا الْبَلَدِ⦿
352.	وَ اَنْتَ حِلٌّۢ بِهٰذَا الْبَلَدِ⦿
353.	وَ وَالِدٍ وَّ مَا وَلَدَ⦿
354.	لَقَدْ خَلَقْنَا الْاِنْسَانَ فِیْ كَبَدٍ⦿
355.	اَیَحْسَبُ اَنْ لَّنْ یَّقْدِرَ عَلَیْهِ اَحَدٌ⦿
356.	یَقُوْلُ اَهْلَكْتُ مَالًا لُّبَدًا⦿
357.	اَیَحْسَبُ اَنْ لَّمْ یَرَهٗۤ اَحَدٌ⦿
358.	اَلَمْ نَجْعَلْ لَّهٗ عَیْنَیْنِ⦿
359.	وَ لِسَانًا وَّ شَفَتَیْنِ⦿
360.	وَ هَدَیْنٰهُ النَّجْدَیْنِ⦿
361.	فَلَا اقْتَحَمَ الْعَقَبَةَ⦿
362.	وَ مَاۤ اَدْرٰىكَ مَا الْعَقَبَةُ⦿
363.	فَكُّ رَقَبَةٍ⦿
364.	اَوْ اِطْعٰمٌ فِیْ یَوْمٍ ذِیْ مَسْغَبَةٍ⦿
365.	یَّتِیْمًا ذَا مَقْرَبَةٍ⦿
366.	اَوْ مِسْكِیْنًا ذَا مَتْرَبَةٍ⦿
367.	ثُمَّ كَانَ مِنَ الَّذِیْنَ اٰمَنُوْا وَ تَوَاصَوْا بِالصَّبْرِ وَ تَوَاصَوْا بِالْمَرْحَمَةِ⦿
368.	اُولٰٓىٕكَ اَصْحٰبُ الْمَیْمَنَةِ⦿
369.	وَ الَّذِیْنَ كَفَرُوْا بِاٰیٰتِنَا هُمْ اَصْحٰبُ الْمَشْــٴَـمَةِ⦿
370.	عَلَیْهِمْ نَارٌ مُّؤْصَدَةٌ⦿
سُورَۃُ الشَّمسِ
371.	وَ الشَّمْسِ وَ ضُحٰىهَا⦿
372.	وَ الْقَمَرِ اِذَا تَلٰىهَا⦿
373.	وَ النَّهَارِ اِذَا جَلّٰىهَا⦿
374.	وَ الَّیْلِ اِذَا یَغْشٰىهَا⦿
375.	وَ السَّمَآءِ وَ مَا بَنٰىهَا⦿
376.	وَ الْاَرْضِ وَ مَا طَحٰىهَا⦿
377.	وَ نَفْسٍ وَّ مَا سَوّٰىهَا⦿
378.	فَاَلْهَمَهَا فُجُوْرَهَا وَ تَقْوٰىهَا⦿
379.	قَدْ اَفْلَحَ مَنْ زَكّٰىهَا⦿
380.	وَ قَدْ خَابَ مَنْ دَسّٰىهَا⦿
381.	كَذَّبَتْ ثَمُوْدُ بِطَغْوٰىهَا⦿
382.	اِذِ انْۢبَعَثَ اَشْقٰىهَا⦿
383.	فَقَالَ لَهُمْ رَسُوْلُ اللّٰهِ نَاقَةَ اللّٰهِ وَ سُقْیٰهَا⦿
384.	فَكَذَّبُوْهُ فَعَقَرُوْهَاﭪ--فَدَمْدَمَ عَلَیْهِمْ رَبُّهُمْ بِذَنْۢبِهِمْ فَسَوّٰىهَا⦿
385.	وَ لَا یَخَافُ عُقْبٰهَا⦿
الَّیلِ
386.	وَ الَّیْلِ اِذَا یَغْشٰى⦿
387.	وَ النَّهَارِ اِذَا تَجَلّٰى⦿
388.	وَ مَا خَلَقَ الذَّكَرَ وَ الْاُنْثٰۤى⦿
389.	اِنَّ سَعْیَكُمْ لَشَتّٰى⦿
390.	فَاَمَّا مَنْ اَعْطٰى وَ اتَّقٰى⦿
391.	وَ صَدَّقَ بِالْحُسْنٰى⦿
392.	فَسَنُیَسِّرُهٗ لِلْیُسْرٰى⦿
393.	وَ اَمَّا مَنْۢ بَخِلَ وَ اسْتَغْنٰى⦿
394.	وَ كَذَّبَ بِالْحُسْنٰى⦿
395.	فَسَنُیَسِّرُهٗ لِلْعُسْرٰى⦿
396.	وَ مَا یُغْنِیْ عَنْهُ مَالُهٗۤ اِذَا تَرَدّٰى⦿
397.	اِنَّ عَلَیْنَا لَلْهُدٰى⦿
398.	وَ اِنَّ لَنَا لَلْاٰخِرَةَ وَ الْاُوْلٰى⦿
399.	فَاَنْذَرْتُكُمْ نَارًا تَلَظّٰى⦿
400.	لَا یَصْلٰىهَاۤ اِلَّا الْاَشْقَى⦿
401.	الَّذِیْ كَذَّبَ وَ تَوَلّٰى⦿
402.	وَ سَیُجَنَّبُهَا الْاَتْقَى⦿
403.	الَّذِیْ یُؤْتِیْ مَالَهٗ یَتَزَكّٰى⦿
404.	وَ مَا لِاَحَدٍ عِنْدَهٗ مِنْ نِّعْمَةٍ تُجْزٰۤى⦿
